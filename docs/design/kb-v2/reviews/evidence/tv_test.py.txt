import tantivy, os, shutil, subprocess, sys, json, time
p = "/tmp/kbv2r/x/tvt"; shutil.rmtree(p, ignore_errors=True); os.makedirs(p)

idx = tantivy.Index(sch, path=p); w = idx.writer()
print("writer methods:", [m for m in dir(w) if not m.startswith("_")])
print("index methods:", [m for m in dir(idx) if not m.startswith("_")])
