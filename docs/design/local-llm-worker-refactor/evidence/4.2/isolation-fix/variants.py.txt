def run(selected):
    out = {}
    for tid in sorted(k for k in REGISTRY if k != "B4-EXP-D"):
        task = REGISTRY[tid]()
        for name in selected:
            ends = gold_ends(task)
            path = task.stages[0].required_change_paths[0]
            for end in ends:
                end[path] = V[name] + end[path]
            extra = task.gold_reports(task.base_files, ends) if task.gold_reports else None
            with materialize(task, ends, reports_for(task, ends, extra), session=task.gold_session) as attempt:
                ev = evaluate_attempt(task, attempt)
            detail = ""
            if ev.verdict != "PASS":
                for s in ev.stages:
                    bad = [c for c in s.checks if c.get("passed") is False]
                    if bad: detail = f"{s.stage_id}:{bad[0]['code']}:{bad[0]['detail'][:90]}"; break
            out.setdefault(name, []).append((tid, ev.verdict, ev.fail_code, detail, ",".join(f for f in ev.flags if f not in ('development_answers_public','sandbox_unavailable','trace_unavailable'))))
    return out

if __name__ == "__main__":
    res = run(list(V))
    summary = {}
    for name, rows in res.items():
        npass = sum(r[1] == "PASS" for r in rows)
        summary[name] = {"pass": npass, "of": len(rows), "fails": [r for r in rows if r[1] != "PASS"][:2], "flags": sorted({r[4] for r in rows if r[4]})}
    json.dump(summary, open(sys.argv[1], "w"), indent=1)
    for name, s in summary.items():
        print(f"{name:34s} {s['pass']}/{s['of']}", (s['fails'][0][2:] if s['fails'] else ''), s['flags'])
